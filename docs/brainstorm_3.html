phi = np.linspace(0, 2*np.pi, 500)

norm = np.sqrt(15/(16*np.pi))

Y = norm * np.cos(2*phi)

y_pos = np.where(Y>0, Y, np.nan)
y_neg = np.where(Y<0, -Y, np.nan)

fig = plt.figure(figsize = ((7,7)))
ax = fig.add_subplot(111, projection = "polar")

ax.plot(phi, y_pos, color = "tab:blue", linewidth = 2, label = "positive lobe")

ax.plot(phi, y_neg, color = "tab:orange", linewidth = 2, label = "negative lobe")

ax.set_title(r"2D Polar Projection at $3d_{x^2-y^2}$ Spherical harmonic", pad = 20)

ax.legend(loc = "upper right")
ax.set_rlim((0, 0.6))

plt.tight_layout()
plt.show()
